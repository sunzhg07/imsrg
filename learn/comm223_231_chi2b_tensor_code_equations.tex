\section*{Implemented Barred Two-body Transform}
For an operator $O$ (used for both $\Omega\equiv\mathrm{Eta}$ and
$\Gamma$ in code), the implemented barred transform is
\begin{align}
\bar O_{ajkb}^{J_0J_1\lambda}
=
\phi\!\left(J_0+\frac{j_a+j_k}{2}+\lambda\right)
\sqrt{(2J_0+1)(2J_1+1)}
\sum_{J_2J_3j_0}
\phi(J_2)
\sqrt{(2J_2+1)(2J_3+1)}(2j_0+1)
\\
\times
\sixj{\lambda}{J_3}{J_2}{j_a/2}{j_j/2}{j_0}
\sixj{j_a/2}{j_b/2}{J_0}{j_k/2}{j_0}{J_3}
\sixj{J_1}{J_0}{\lambda}{j_0}{j_j/2}{j_k/2}
\,O_{abjk}^{J_2J_3\lambda}.
\end{align}

Code loop limits:
\begin{align}
J_2 &\in \left[\frac{|j_a-j_b|}{2},\frac{j_a+j_b}{2}\right],
&
J_3 &\in \left[\frac{|j_j-j_k|}{2},\frac{j_j+j_k}{2}\right],
&
j_0 &\in \left[\frac{|j_a-j_j|}{2},\frac{j_a+j_j}{2}\right],
\end{align}
plus the triangle filter $|J_2-J_3|\le\lambda\le J_2+J_3$.

\section*{Implemented $\bar\chi^{\gamma}$-like Intermediate (IIIa block)}
The helper \texttt{bar\_chi\_gamma(i,l,k,j;J0)} implements
\begin{align}
\bar\chi^{\gamma}_{ilkj}(J_0)
=
\frac{\phi(J_0)}{2J_0+1}
\sum_{abJ_2}
\Big(n_a\bar n_b n_l\bar n_k - \bar n_a n_b\bar n_l n_k\Big)
\phi(J_2+\lambda)
\hat\lambda^{-1}
\bar\Omega_{ibaj}^{J_0J_2\lambda}
\bar\Omega_{alkb}^{J_2J_0\lambda},
\end{align}
with
\begin{align}
J_2\in
\left[
\frac{\max(|j_a-j_j|,|j_a-j_l|)}{2},
\frac{\min(j_a+j_j,j_a+j_l)}{2}
\right],
\end{align}
and $|J_0-J_2|\le\lambda\le J_0+J_2$.

\section*{Implemented Final One-body IIIa + IIIb Update}

\sum_{abcJ_0}(2J_0+1)
\left[
\bar\chi^{\gamma}_{pcab}(J_0)\,\bar\Gamma_{abqc}^{J_0J_0 0}

\sum_{abcJ_0}(2J_0+1)

\left[
