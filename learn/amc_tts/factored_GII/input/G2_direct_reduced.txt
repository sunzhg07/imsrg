# Same as G2_direct.txt but Case-2 reduce=true on Γ and G2.
# Do NOT mix with unreduced Path B / m 2-CG unpack without Ĵ conversion.

declare Omega {
    mode=4,
    latex="\\Omega",
    scalar=false
}

declare Gamma {
    mode=4,
    latex="\\Gamma",
    scalar=true,
    reduce=true
}

declare n {
    mode=2,
    diagonal=true,
    latex="n"
}

declare nbar {
    mode=2,
    diagonal=true,
    latex="\\bar{n}"
}

declare G2 {
    mode=4,
    latex="\\Gamma^{(II)}",
    scalar=true,
    reduce=true
}

G2_ijkl = 1/2 * sum_abcd((n_a*n_b*nbar_c + nbar_a*nbar_b*n_c)
    * (P(i/j) * Gamma_ajbc * Omega_bcad * Omega_idkl
     - P(k/l) * Gamma_adbc * Omega_bcak * Omega_ijdl));
