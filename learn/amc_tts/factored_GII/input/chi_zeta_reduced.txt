# χ^ζ with Γ reduce=true (Case-2). χ stays tensor (AMC default reduced for non-scalar).

declare Omega {
    mode=4,
    latex="\\Omega",
    scalar=false
}

declare Gamma {
    mode=4,
    latex="\\Gamma",
    scalar=true,
    reduce=true
}

declare n {
    mode=2,
    diagonal=true,
    latex="n"
}

declare nbar {
    mode=2,
    diagonal=true,
    latex="\\bar{n}"
}

declare chizeta {
    mode=2,
    scalar=false,
    latex="\\chi^{\\zeta}"
}

chizeta_ij = 1/2 * sum_abc((n_a*n_b*nbar_c + nbar_a*nbar_b*n_c) * Gamma_aibc * Omega_bcaj);
