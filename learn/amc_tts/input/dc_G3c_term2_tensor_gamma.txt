declare Omega {
    mode=4,
    latex="\Omega",
    scalar=false
}

declare Gamma {
    mode=4,
    latex="\Gamma",
    scalar=false
}

declare n {
    mode=2,
    diagonal=true,
    latex="n"
}

declare nbar {
    mode=2,
    diagonal=true,
    latex="\bar{n}"
}

declare G3c_T2 {
    mode=4,
    latex="\Gamma^{(III_c)}_{T2}",
    scalar=false
}

# Term2 only; Omega & Gamma both tensors (reduced MEs)
G3c_T2_ijkl = -1/2 * sum_abcd(
    (nbar_a*nbar_b*n_c + n_a*n_b*nbar_c)
    * P(i/j) * P(k/l)
    * Omega_icab*Omega_abdl*Gamma_djkc);
