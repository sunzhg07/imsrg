declare Omega {
    mode=4,
    latex="\Omega",
    scalar=false
}

declare Gamma {
    mode=4,
    latex="\Gamma",
    scalar=false
}

declare n {
    mode=2,
    diagonal=true,
    latex="n"
}

declare nbar {
    mode=2,
    diagonal=true,
    latex="\bar{n}"
}

declare G3c {
    mode=4,
    latex="\Gamma^{(III_c)}",
    scalar=false
}

# Both Omega and Gamma declared as tensors → AMC uses reduced MEs for both.
# (Scalar Gamma default would be unreduced and mix conventions with tensor Omega.)
G3c_ijkl = -1/2 * sum_abcd(
    (nbar_a*nbar_b*n_c + n_a*n_b*nbar_c)
    * P(i/j) * P(k/l)
    * (Omega_abcl*Omega_idab*Gamma_cjkd + Omega_icab*Omega_abdl*Gamma_djkc));
