# Direct only. Expand of Path B (locked ≡ m):
#   G2 = 1/2 Σ w [P(i/j) Γ_ajbc Ω_bcad Ω_idkl − P(k/l) Γ_adbc Ω_bcak Ω_ijdl]
# See factored_GII/input/G2_direct.txt. Unreduced Γ/G2.

declare Omega {
    mode=4,
    latex="\\Omega",
    scalar=false
}

declare Gamma {
    mode=4,
    latex="\\Gamma",
    scalar=true
}

declare n {
    mode=2,
    diagonal=true,
    latex="n"
}

declare nbar {
    mode=2,
    diagonal=true,
    latex="\\bar{n}"
}

declare G2 { mode=4, latex="\Gamma^{(II)}", scalar=true }

G2_ijkl = 1/2 * sum_abcd((n_a*n_b*nbar_c + nbar_a*nbar_b*n_c)
    * (P(i/j) * Gamma_ajbc * Omega_bcad * Omega_idkl
     - P(k/l) * Gamma_adbc * Omega_bcak * Omega_ijdl));
