    diagonal=true,

    mode=2,

    latex="\\bar{n}"

declare chiepsilon {
    mode=2,
    scalar=true,
    reduce=true,
    latex="\\chi^{\\epsilon}"
}

chiepsilon_ij = 1/2 * sum_abc((nbar_a*nbar_b*n_c + n_a*n_b*nbar_c) * Omega_ciab * Omega_abcj);
