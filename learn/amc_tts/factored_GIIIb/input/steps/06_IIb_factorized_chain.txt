# Full scalar IIb factorized chain (m-scheme steps AMC can reduce)
# 1) Pandya Gamma
# 2) Pandya Omega → barChi
# 3) RC barChi → dbarChi (skeleton, no swap yet)
# 4) GEMM W_jlik = Σ barG_jlab * dbarChi_abik
# 5) InvPandya onto Z_jikl = -W_ijkl  (LHS jikl encodes modified external layout)
#    Restore (1-P_ij)(1-P_kl) by hand as in code.

declare Omega { mode=4, latex="\\Omega" }
declare Gamma { mode=4, latex="\\Gamma" }

declare barO {
    mode=4, latex="\\bar{\\Omega}", scheme=((1,-4),(3,-2))
}
declare barG {
    mode=4, latex="\\bar{\\Gamma}", scheme=((1,-4),(3,-2))
}
barO_ijkl = - Omega_ijkl;
barG_ijkl = - Gamma_ijkl;

declare n { mode=2, diagonal=true, latex="n" }
declare nbar { mode=2, diagonal=true, latex="\\bar{n}" }

declare barChi {
    mode=4, latex="\\bar{\\chi}^{\\eta}", scalar=true, scheme=((1,-4),(3,-2))
}
barChi_ijkl = sum_ab(
    (nbar_a*n_b*nbar_k + n_a*nbar_b*n_k) * (barO_iabl * barO_bjka)
);

declare dbarChi {
    mode=4, latex="\\overline{\\overline{\\chi}}^{\\eta}", scalar=true, scheme=((1,-3),(4,-2))
}
# Code: XbarIIbd -= ... so overall minus on RC skeleton piece
dbarChi_ijkl = - barChi_ijkl;

declare W {
    mode=4, latex="W", scheme=((1,-4),(3,-2))
}
W_jlik = sum_ab(barG_jlab * dbarChi_abik);

declare Z {
    mode=4, latex="Z^{\\mathrm{IIb}}"
}
# Modified InvPandya layout: print formula for Z at jikl from W at ijkl
Z_jikl = - W_ijkl;
