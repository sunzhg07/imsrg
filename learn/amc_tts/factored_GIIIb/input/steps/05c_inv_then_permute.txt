# Modified InvPandya = standard InvPandya composed with index permute on Z
# Code reads W_{j l̄, i k̄} for Z_{ijkl} path; equivalent idea:
#   Zstd = InvPandya(W) with Zstd_ijkl <-> W_{i l̄ k j̄}
#   If we need W_{j l̄ i k̄}, that is Zstd with labels j,l,i,k
#   so Z_ijkl = Zstd_jlik
declare W {
    mode=4, latex="W", scheme=((1,-4),(3,-2))
}
declare Zstd {
    mode=4, latex="Z^{\\mathrm{std}}"
}
Zstd_ijkl = - W_ijkl;

declare Z {
    mode=4, latex="Z^{\\mathrm{IIb}}"
}
Z_ijkl = Zstd_jlik;
