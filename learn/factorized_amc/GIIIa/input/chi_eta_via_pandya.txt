# χ^η Path B: Pandya(Ω) → product → inv Pandya
# Same m-scheme as chi_eta_direct.txt
#
# AMC sample often uses chi = -barChi → Path B = −direct/m.
# Corrected: chi = +barChi (tts_ring / chi_gamma / OMEGA_TT_TO_SCALAR rule).
declare Omega {
    mode=4,
    latex="\\Omega",
    scalar=false
}
declare barO {
    mode=4,
    latex="\\bar{\\Omega}",
    scalar=false,
    scheme=((1,-4),(3,-2))
}
barO_ijkl = - Omega_ijkl;

declare n { mode=2, diagonal=true, latex="n" }
declare nbar { mode=2, diagonal=true, latex="\\bar{n}" }

declare barChi {
    mode=4,
    latex="\\bar{\\chi}^{\\eta}",
    scalar=true,
    reduce=true,
    scheme=((1,-4),(3,-2))
}
barChi_ijkl = sum_ab((nbar_a*n_b*nbar_k + n_a*nbar_b*n_k) * (barO_iabl * barO_bjka));

declare chi {
    mode=4,
    latex="\\chi^{\\eta}",
    scalar=true,
    reduce=true
}
chi_ijkl = barChi_ijkl;
