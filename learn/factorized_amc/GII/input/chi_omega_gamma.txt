# AMC index order mirrors chi_zeta (Ω_aibc Γ_bcaj).

    latex="\\bar{n}"

declare chiOG {
    mode=2,
    scalar=false,
    latex="\\chi^{\\Omega\\Gamma}"
}

chiOG_ij = 1/2 * sum_abc((n_a*n_b*nbar_c + nbar_a*nbar_b*n_c) * Omega_aibc * Gamma_bcaj);
