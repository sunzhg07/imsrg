# m-scheme: χ_ij(mi,mj) = 1/2 Σ_abc w Γ_aibc Ω_bcaj  (no extra [×]_0; rank follows Ω)

declare Omega {
    mode=4,
    latex="\\Omega",
    scalar=false
}

declare Gamma {
    mode=4,
    latex="\\Gamma",
    scalar=true
}

declare n {
    mode=2,
    diagonal=true,
    latex="n"
}

declare nbar {
    mode=2,
    diagonal=true,
    latex="\\bar{n}"
}

declare chizeta {
    mode=2,
    scalar=false,
    latex="\\chi^{\\zeta}"
}

chizeta_ij = 1/2 * sum_abc((n_a*n_b*nbar_c + nbar_a*nbar_b*n_c) * Gamma_aibc * Omega_bcaj);
