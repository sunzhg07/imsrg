declare Omega {
    mode=4,
    latex="\\Omega",
    scalar=false
}

declare Gamma {
    mode=4,
    latex="\\Gamma",
    scalar=true
}

declare n {
    mode=2,
    diagonal=true,
    latex="n"
}

declare nbar {
    mode=2,
    diagonal=true,
    latex="\\bar{n}"
}

declare Chi {
    mode=4,
    latex="\\chi^{\\kappa}",
    scalar=false,
    reduce=true
}

Chi_ijbd = sum_ac(
    (nbar_c*nbar_d*n_a + n_c*n_d*nbar_a) * Omega_aicd * Gamma_jcba);
