# χ^λ Term 1 only: T×S → tensor. w_l Γ_ijab Ω_abkl
# Γ unreduced scalar; Ω/χ reduced tensors.

declare Omega {
    mode=4,
    latex="\\Omega",
    scalar=false
}

declare Gamma {
    mode=4,
    latex="\\Gamma",
    scalar=true
}

declare n {
    mode=2,
    diagonal=true,
    latex="n"
}

declare nbar {
    mode=2,
    diagonal=true,
    latex="\\bar{n}"
}

declare Chi {
    mode=4,
    latex="\\chi^{\\lambda}",
    scalar=false
}

Chi_ijkl = sum_ab(
    (nbar_a*nbar_b*n_l + n_a*n_b*nbar_l) * Gamma_ijab * Omega_abkl);
