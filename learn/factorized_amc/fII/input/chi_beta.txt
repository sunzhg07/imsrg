# From unfactored f^II string Gamma_cdab Omega_abce (occ on free e).
# Overall 1/2 kept here; f^II contraction should NOT repeat it.
chi_de = 1/2 * sum_abc((n_a*n_b*nbar_c*nbar_e - nbar_a*nbar_b*n_c*n_e) * (Gamma_cdab*Omega_abce));
